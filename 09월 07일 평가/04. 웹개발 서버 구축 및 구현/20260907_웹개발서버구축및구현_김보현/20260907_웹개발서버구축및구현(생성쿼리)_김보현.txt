#차량 판매 정보 시스템

#자동차정보(CAR_INFO)
CREATE TABLE CAR_INFO (
	MODEL_NUM INT PRIMARY KEY AUTO_INCREMENT
	, MODEL_NAME VARCHAR(30) NOT NULL
	, CAR_PRICE INT NOT NULL
	, PRODUCT_COM VARCHAR(30) NOT NULL
);

#자동차판매정보(SALES_INFO)
CREATE TABLE SALES_INFO (
	SALES_NUM INT PRIMARY KEY AUTO_INCREMENT
	, OWNER_NAME VARCHAR(30) NOT NULL
	, OWNER_TEL VARCHAR(15) #010-1111-1111
	, COLOR VARCHAR(20) NOT NULL
	, SALES_DATE DATETIME NOT NULL DEFAULT SYSDATE()
	, MODEL_NUM INT
	, CONSTRAINT FK_MODEL_NUM FOREIGN KEY (MODEL_NUM) REFERENCES CAR_INFO (MODEL_NUM)
);
